当你开始写给别人用的组件（组件库、团队公共组件）时，API 设计就和实现一样重要。好的组件 API 应该**灵活、直观、难以误用**。

## 模式一：复合组件（Compound Components）

一组组件协同工作，共享隐式的状态，就像 HTML 的 `<select>` 和 `<option>`。使用者可以自由组合结构，而不用传一大堆配置 props。

```jsx play title="复合组件：Tabs"
import { createContext, useContext, useState } from 'react';

const TabsContext = createContext(null);

function Tabs({ defaultValue, children }) {
  const [active, setActive] = useState(defaultValue);
  return <TabsContext value={{ active, setActive }}>{children}</TabsContext>;
}

function TabList({ children }) {
  return <div role="tablist" style={{ display: 'flex', gap: 4, borderBottom: '2px solid #ddd' }}>{children}</div>;
}

function Tab({ value, children }) {
  const { active, setActive } = useContext(TabsContext);
  const selected = active === value;
  return (
    <button role="tab" aria-selected={selected} onClick={() => setActive(value)}
      style={{ border: 0, padding: '6px 12px', background: selected ? '#087ea4' : 'transparent', color: selected ? '#fff' : 'inherit', borderRadius: '6px 6px 0 0' }}>
      {children}
    </button>
  );
}

function TabPanel({ value, children }) {
  const { active } = useContext(TabsContext);
  return active === value ? <div role="tabpanel" style={{ padding: 10 }}>{children}</div> : null;
}

// 挂载为静态属性，使用时更有“一家人”的感觉
Tabs.List = TabList; Tabs.Tab = Tab; Tabs.Panel = TabPanel;

function App() {
  return (
    <Tabs defaultValue="intro">
      <Tabs.List>
        <Tabs.Tab value="intro">简介</Tabs.Tab>
        <Tabs.Tab value="spec">规格</Tabs.Tab>
        <Tabs.Tab value="review">评价 ⭐</Tabs.Tab>
      </Tabs.List>
      <Tabs.Panel value="intro">这是一款机械键盘。</Tabs.Panel>
      <Tabs.Panel value="spec">87 键，热插拔，三模连接。</Tabs.Panel>
      <Tabs.Panel value="review">“手感很好！”</Tabs.Panel>
    </Tabs>
  );
}
```

上面的 `Tabs` 把 `TabsContext` 直接当 Provider 用：`<TabsContext value={…}>`。Context 的用法见 [Context：跨层级共享数据](/lessons/context)。

<CallBox kind="tip" label="维护 React 18 项目时">

React 18 要写 `<TabsContext.Provider value={…}>`。React 19 也还支持这种写法，旧代码不用急着改。

</CallBox>

<CallBox kind="warn" label="常见坑">

如果项目用了服务端组件（见 [Server Components 与全栈 React](/lessons/server-components)），要小心 `Tabs.List` 这种静态属性写法。`Tabs` 用了 `useState`，必须是客户端组件。在**服务端组件**里写 `<Tabs.List>` 会报错，因为服务端拿到的只是客户端模块的引用，不能再取它的属性。这时改成具名导出：`export { Tabs, TabList, Tab, TabPanel }`。Radix 的 `Tabs.Root`、`Tabs.List` 是整个模块的命名空间导入，不是组件上的静态属性，所以不受影响。

</CallBox>

## 模式二：受控与非受控双模式

好的组件支持两种模式。非受控模式：组件自己管理 state，开箱即用。受控模式：父组件管理 state，可以完全定制。判断依据是：使用者是否传入了 `value`。

```jsx title="原生 <input> 的 value / defaultValue 就是这种设计"
function Toggle({ value, defaultValue = false, onChange }) {
  const [inner, setInner] = useState(defaultValue);
  const isControlled = value !== undefined;
  const on = isControlled ? value : inner;

  function toggle() {
    if (!isControlled) setInner(!on);
    onChange?.(!on);
  }
  return <button onClick={toggle}>{on ? '开' : '关'}</button>;
}

<Toggle defaultValue={true} />                 // 非受控：组件自己管
<Toggle value={on} onChange={setOn} />        // 受控：父组件说了算
```

## 模式三：Render Props 与 HOC（了解历史）

在 Hooks 出现之前（2019 年前），复用状态逻辑主要靠这两种模式。你仍会在老代码和一些库中看到它们：

- **Render Props**：组件接收一个返回 JSX 的函数 prop，自己管 state，把数据交给这个函数去渲染。
- **高阶组件（HOC）**：一个函数，接收一个组件，返回包着 state 的新组件，用 props 把数据注入给原组件。

下面用同一段逻辑（跟踪鼠标位置）写三遍。先看懂前两种的实现，再看自定义 Hook 怎样把它们简化。

```jsx play title="同一段逻辑的三种写法"
import { useState } from 'react';

const area = { border: '1px solid #bbb', padding: 8, marginBottom: 8 };

// 写法一：HOC。接收组件，返回包着 state 的新组件，用 props 注入 pos
function withMouse(Component) {
  return function WithMouse(props) {
    const [p, setP] = useState({ x: 0, y: 0 });
    return (
      <div style={area} onMouseMove={e => setP({ x: e.clientX, y: e.clientY })}>
        <Component {...props} pos={p.x + ',' + p.y} />
      </div>
    );
  };
}
// 另一个 HOC，也注入名为 pos 的 prop
function withOrigin(Component) {
  return function WithOrigin(props) {
    return <Component {...props} pos="原点" />;
  };
}
function ShowPos({ pos }) {
  return <p>HOC：{pos}</p>;
}
const HocDemo = withMouse(withOrigin(ShowPos));

// 写法二：Render Props。组件管 state，把数据交给 render 函数
function Mouse({ render }) {
  const [p, setP] = useState({ x: 0, y: 0 });
  return (
    <div style={area} onMouseMove={e => setP({ x: e.clientX, y: e.clientY })}>
      {render(p)}
    </div>
  );
}

// 写法三：自定义 Hook。直接返回值，名字由调用者决定
function useMouse() {
  const [p, setP] = useState({ x: 0, y: 0 });
  const bind = { onMouseMove: e => setP({ x: e.clientX, y: e.clientY }) };
  return { x: p.x, y: p.y, bind };
}
function HookDemo() {
  const { x, y, bind } = useMouse();
  return (
    <div style={area} {...bind}>
      <p>Hook：{x},{y}</p>
    </div>
  );
}

function App() {
  return (
    <div>
      <p>把鼠标移到下面的框里：</p>
      <HocDemo />
      <Mouse render={({ x, y }) => <p>Render Props：{x},{y}</p>} />
      <HookDemo />
    </div>
  );
}
```

HOC 有一个固有的毛病：被包的组件看不出 `pos` 是哪个 HOC 注入的。两个 HOC 注入同名 prop 时，只有一个能留下来，另一个悄悄丢失，没有任何警告。Render Props 没有这个问题，但嵌套多了就是一层套一层的回调。自定义 Hook 两个毛病都没有。

HOC 还有一个小麻烦：`ref` 要不要传给被包的组件。React 19 里，函数组件的 `ref` 是普通 prop，HOC 里的 `{...props}` 会把它一起传下去。React 18 里 `ref` 不是 prop，HOC 要用 `forwardRef` 才能转发。

<CallBox kind="tip" label="要点">

Render Props 现在仍然有用。如果组件要让**使用者决定如何渲染内部数据**，就用它，例如虚拟列表的 `renderItem={(item) => ...}`。

</CallBox>

## 其他设计原则

- **组合优于配置**：不要提供 `showIcon`、`iconPosition`、`iconColor` 这样一长串 props。让使用者传 children 或插槽。
- **透传原生属性**：`function Button({ variant, ...rest }) { return <button {...rest} /> }`，让使用者能传 aria-\*、onClick、type 等。React 19 里 `ref` 也是普通 prop，所以 `...rest` 会把 `ref` 一起透传，不需要 `forwardRef`。
- **无障碍**：用正确的语义标签和 ARIA 角色，支持键盘操作。
